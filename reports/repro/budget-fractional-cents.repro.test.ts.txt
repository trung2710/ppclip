// Tái hiện lỗi: cost dạng số lẻ (double precision) nhưng bảng budget_incidents vẫn là integer.
//
// Cách chạy:
//   1. Sao chép file này thành server/src/__tests__/zz-tmp-budget-fractional-repro.test.ts (bỏ đuôi .txt)
//   2. npx vitest run server/src/__tests__/zz-tmp-budget-fractional-repro.test.ts
//   3. Xóa file sau khi chạy.
//
// Kết quả khi viết báo cáo (29/10/2026, commit f6c9d6bfc): cả 4 test đều PASS, trong đó 2 ca "fractional"
// PASS vì evaluateCostEvent ném lỗi `invalid input syntax for type integer: "80.5"` / `"100.37"`.
// Khi đã sửa lỗi, 2 ca "fractional" sẽ FAIL; lúc đó đổi expect thành "no error" và status "paused".

import { randomUUID } from "node:crypto";
import { afterAll, afterEach, beforeAll, describe, expect, it, vi } from "vitest";
import {
  agents,
  approvals,
  budgetIncidents,
  budgetPolicies,
  companies,
  costEvents,
  createDb,
} from "@paperclipai/db";
import { eq } from "drizzle-orm";
import { budgetService } from "../services/budgets.ts";
import {
  getEmbeddedPostgresTestSupport,
  startEmbeddedPostgresTestDatabase,
} from "./helpers/embedded-postgres.js";

vi.mock("../services/activity-log.js", () => ({ logActivity: vi.fn() }));

const support = await getEmbeddedPostgresTestSupport();
const describeDb = support.supported ? describe : describe.skip;

describeDb("TEMP repro: fractional cents vs integer budget columns", () => {
  let db!: ReturnType<typeof createDb>;
  let tempDb: Awaited<ReturnType<typeof startEmbeddedPostgresTestDatabase>> | null = null;

  beforeAll(async () => {
    tempDb = await startEmbeddedPostgresTestDatabase("paperclip-tmp-fractional-");
    db = createDb(tempDb.connectionString);
  }, 60_000);

  afterEach(async () => {
    await db.delete(budgetIncidents);
    await db.delete(approvals);
    await db.delete(budgetPolicies);
    await db.delete(costEvents);
    await db.delete(agents);
    await db.delete(companies);
  });

  afterAll(async () => {
    await tempDb?.cleanup();
  });

  async function fixture(costCents: number) {
    const companyId = randomUUID();
    const agentId = randomUUID();
    await db.insert(companies).values({
      id: companyId,
      name: "Paperclip",
      issuePrefix: `T${companyId.replace(/-/g, "").slice(0, 6).toUpperCase()}`,
      requireBoardApprovalForNewAgents: false,
    });
    await db.insert(agents).values({
      id: agentId,
      companyId,
      name: "Fractional Agent",
      role: "engineer",
      status: "active",
      adapterType: "codex_local",
      adapterConfig: {},
      runtimeConfig: {},
      permissions: {},
    });
    await db.insert(budgetPolicies).values({
      companyId,
      scopeType: "agent",
      scopeId: agentId,
      metric: "billed_cents",
      windowKind: "calendar_month_utc",
      amount: 100, // 100 cent = $1.00
      warnPercent: 80,
      hardStopEnabled: true,
      notifyEnabled: true,
      isActive: true,
    });
    const [event] = await db
      .insert(costEvents)
      .values({
        companyId,
        agentId,
        provider: "openai",
        biller: "litellm",
        billingType: "metered_api",
        model: "gemini/gemini-3.6-flash",
        inputTokens: 100,
        cachedInputTokens: 10,
        outputTokens: 20,
        costCents,
        occurredAt: new Date(),
      })
      .returning();
    return { companyId, agentId, event: event! };
  }

  async function run(label: string, costCents: number) {
    const { agentId, event } = await fixture(costCents);
    const cancelWorkForScope = vi.fn().mockResolvedValue(undefined);
    const service = budgetService(db, { cancelWorkForScope });
    let outcome = "no error";
    try {
      await service.evaluateCostEvent(event);
    } catch (e) {
      outcome = `ERROR: ${(e as Error).message.split("\n")[0]}`;
      const cause = (e as { cause?: { message?: string } }).cause?.message;
      if (cause) outcome += ` | cause: ${cause}`;
    }
    const [agent] = await db.select({ status: agents.status }).from(agents).where(eq(agents.id, agentId));
    const incidents = await db
      .select({ t: budgetIncidents.thresholdType, obs: budgetIncidents.amountObserved })
      .from(budgetIncidents);
    const approvalRows = await db.select({ id: approvals.id }).from(approvals);
    // stderr: cấu hình vitest của repo ẩn console.log
    process.stderr.write(
      `\n[REPRO] ${label} (cost ${costCents}c, limit 100c): ${outcome}; agent.status=${agent?.status}; ` +
        `incidents=${JSON.stringify(incidents)}; approvals=${approvalRows.length}\n`,
    );
    return { outcome, status: agent?.status };
  }

  it("whole cents, soft threshold (control)", async () => {
    const r = await run("soft, whole", 80);
    expect(r.outcome).toBe("no error");
  });

  it("fractional cents, soft threshold", async () => {
    const r = await run("soft, fractional", 80.5);
    expect(r.outcome).toContain("ERROR");
  });

  it("whole cents, hard stop (control)", async () => {
    const r = await run("hard, whole", 101);
    expect(r.status).toBe("paused");
  });

  it("fractional cents, hard stop", async () => {
    const r = await run("hard, fractional", 100.37);
    expect(r.outcome).toContain("ERROR");
  });
});
